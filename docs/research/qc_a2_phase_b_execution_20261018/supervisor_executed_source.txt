"""Owned-process supervisor: one launch, no retry, no scientific cap/patch."""
from pathlib import Path
import datetime,hashlib,json,os,subprocess,sys,time

ROOT=Path('/tmp/prt-qc-a2-phase-b-execution-20261018')
CANONICAL=Path('/tmp/prt-qc-a2-pilot-phase-a-20261018')
ART=ROOT/'artifacts/qc_a2_phase_b_execution/2026-10-18'
C=ROOT/'docs/research/qc_a2_phase_b_contract_adoption_20261018/execution_contract_C.json'
PYTHON='/home/abe/.pyenv/versions/3.11.1/bin/python'
def now():return datetime.datetime.now(datetime.timezone.utc).isoformat()
def dump(path,data):path.write_text(json.dumps(data,ensure_ascii=False,indent=2)+'\n')
def observe(pid,ledger):
    state={'utc':now(),'pid_in_own_process_namespace':pid,
           'saved_ledger_files':[{'name':p.name,'bytes':p.stat().st_size} for p in sorted(ledger.glob('*')) if p.is_file()],
           'memory_available_kB':next(line.split()[1] for line in Path('/proc/meminfo').read_text().splitlines() if line.startswith('MemAvailable:'))}
    proc=Path('/proc')/str(pid)
    if proc.exists():
        state['process_status']={k:v.strip() for k,v in (line.split(':',1) for line in (proc/'status').read_text().splitlines() if ':' in line)
                                 if k in ('State','VmRSS','VmHWM','VmSize','Threads')}
        state['process_stat']= (proc/'stat').read_text().strip()
    return state
def main():
    pins=json.loads((ART/'external_pins.json').read_text());c=json.loads(C.read_text())
    ledger=CANONICAL/c['execution_store']['ledger_relative']/pins['contract_sha256']
    assert not ledger.exists(),'Existing claim MUST NOT be reused'
    assert hashlib.sha256(C.read_bytes()).hexdigest()==pins['contract_sha256']
    assert hashlib.sha256((ART/'authorization.json').read_bytes()).hexdigest()==pins['authorization_sha256']
    for p in (ART/'launcher_stdout.json',ART/'launcher_stderr.log',ART/'supervisor_start.json',ART/'supervisor_final.json'):
        assert not p.exists(),p
    args=[PYTHON,'-B',str(CANONICAL/'scripts/run_qc_a2_fixed_df_pilot.py'),
          '--repository',str(CANONICAL),'--contract',str(C),'--authorization',str(ART/'authorization.json'),
          '--contract-sha256',pins['contract_sha256'],'--authorization-sha256',pins['authorization_sha256'],
          '--source-commit',pins['source_commit'],'--input-manifest-id',pins['input_manifest_id'],'--execute']
    start=time.monotonic()
    with (ART/'launcher_stdout.json').open('xb') as out,(ART/'launcher_stderr.log').open('xb') as err:
        proc=subprocess.Popen(args,cwd=CANONICAL,stdout=out,stderr=err)
        dump(ART/'supervisor_start.json',{'started_at_utc':now(),'argv':args,'PID_in_supervisor_namespace':proc.pid,
            'supervisor_PID':os.getpid(),'authorization_commit':subprocess.check_output(['git','rev-parse','HEAD'],cwd=ROOT).decode().strip(),
            'no_timeout_or_retry_loop':True,'scientific_source_patched':False})
        print(json.dumps({'status':'SCIENTIFIC_EXECUTION_STARTED','PID':proc.pid,'started_at_utc':now()}),flush=True)
        with (ART/'resource_observations.jsonl').open('x') as log:
            next_print=0
            while proc.poll() is None:
                obs=observe(proc.pid,ledger);obs['elapsed_supervisor_seconds']=time.monotonic()-start
                log.write(json.dumps(obs,ensure_ascii=False)+'\n');log.flush()
                if time.monotonic()>=next_print:
                    print(json.dumps({'status':'RUNNING','elapsed_seconds':round(time.monotonic()-start,1),
                        'own_process':obs.get('process_status'),'saved_files':obs['saved_ledger_files']}),flush=True)
                    next_print=time.monotonic()+40
                time.sleep(5)
    result={'completed_at_utc':now(),'launcher_exit_code':proc.returncode,'supervisor_wall_seconds':time.monotonic()-start,
        'ledger_exists':ledger.exists(),'claim_exists':(ledger/'claim.json').exists(),
        'no_retry':True,'mandatory_STOP':True}
    if (ART/'launcher_stdout.json').stat().st_size:
        try:
            d=json.loads((ART/'launcher_stdout.json').read_text());result.update(scientific_status=d.get('status'),
                failure=d.get('failure'),calls=d.get('calls'),input_opened=d.get('input_opened'))
        except Exception as exc:result['stdout_parse_error']=str(exc)
    dump(ART/'supervisor_final.json',result);print(json.dumps(result),flush=True)
    return 0
if __name__=='__main__':raise SystemExit(main())
